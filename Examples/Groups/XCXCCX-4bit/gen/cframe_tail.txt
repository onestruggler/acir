------------------------------------------------------------------------
-- The frame lemma

private
  ∧ˡ : ∀ {a b} → (a ∧ b) ≡ true → a ≡ true
  ∧ˡ {true} _ = Eq.refl

  ∧ʳ : ∀ {a b} → (a ∧ b) ≡ true → b ≡ true
  ∧ʳ {true} q = q

  ∧-true : ∀ {a b} → a ≡ true → b ≡ true → (a ∧ b) ≡ true
  ∧-true Eq.refl q = q

  false≢true : false ≢ true
  false≢true ()

  refl-false : (u : Fin 4) → (u ==₄ u) ≡ false → ⊥
  refl-false ₀ ()
  refl-false ₁ ()
  refl-false ₂ ()
  refl-false ₃ ()

  flip₁ : ∀ {a b} → a ≡ false → b ≡ true → a ≡ not b
  flip₁ Eq.refl Eq.refl = Eq.refl

  flip₂ : ∀ {a b} → a ≡ true → b ≡ false → a ≡ not b
  flip₂ Eq.refl Eq.refl = Eq.refl

  -- Two positions in the same pair are equal or partners.
  same-half : (y y′ : Fin 8) → half y ≡ half y′ → (y ≡ y′) ⊎ (y ≡ μ₀ y′)
  same-half y y′ e = aux (low y) (low y′) Eq.refl Eq.refl
    where
    aux : (a b : Bool) → low y ≡ a → low y′ ≡ b → (y ≡ y′) ⊎ (y ≡ μ₀ y′)
    aux false false ly ly′ = inj₁ (Eq.trans (Eq.sym (mk-hl y)) (Eq.trans (Eq.cong₂ mk e (Eq.trans ly (Eq.sym ly′))) (mk-hl y′)))
    aux true  true  ly ly′ = inj₁ (Eq.trans (Eq.sym (mk-hl y)) (Eq.trans (Eq.cong₂ mk e (Eq.trans ly (Eq.sym ly′))) (mk-hl y′)))
    aux false true  ly ly′ = inj₂ (Eq.trans (Eq.sym (mk-hl y)) (Eq.cong₂ mk e (flip₁ ly ly′)))
    aux true  false ly ly′ = inj₂ (Eq.trans (Eq.sym (mk-hl y)) (Eq.cong₂ mk e (flip₂ ly ly′)))

  act-inj : (r : CW) {x x′ : Fin 8} → act r x ≡ act r x′ → x ≡ x′
  act-inj r {x} {x′} e = Eq.trans (Eq.sym (act-inv r x)) (Eq.trans (Eq.cong (act (inv r)) e) (act-inv r x′))

module _ (r : CW) (h : InC r) where

  private
    π : Fin 8 → Fin 8
    π = act r

  σᵣ : Fin 4 → Fin 4
  σᵣ v = half (π (mk v false))

  βᵣ : Fin 4 → Bool
  βᵣ v = low (π (mk v false))

  σᵣ-inj : (v v′ : Fin 4) → σᵣ v ≡ σᵣ v′ → v ≡ v′
  σᵣ-inj v v′ e with same-half (π (mk v false)) (π (mk v′ false)) e
  ... | inj₁ p = Eq.trans (Eq.sym (half-mk v false)) (Eq.trans (Eq.cong half (act-inj r p)) (half-mk v′ false))
  ... | inj₂ p = ⊥-elim (false≢true (Eq.trans (Eq.sym (low-mk v false))
                   (Eq.trans (Eq.cong low (act-inj r (Eq.trans p (Eq.trans (Eq.sym (h (mk v′ false)))
                     (Eq.cong π (μ₀-mk v′)))))) (low-mk v′ true))))

  private
    ne′ : (v v′ : Fin 4) → (v ==₄ v′) ≡ false → (b : Bool) → (σᵣ v ==₄ σᵣ v′) ≡ b → not b ≡ true
    ne′ v v′ d false s = Eq.refl
    ne′ v v′ d true  s =
      ⊥-elim (refl-false v (Eq.subst (λ z → (v ==₄ z) ≡ false) (Eq.sym (σᵣ-inj v v′ (==₄-sound _ _ s))) d))

    ne : (v v′ : Fin 4) → (v ==₄ v′) ≡ false → not (σᵣ v ==₄ σᵣ v′) ≡ true
    ne v v′ d = ne′ v v′ d (σᵣ v ==₄ σᵣ v′) Eq.refl

  σᵣ-valid : valid4 (σᵣ ₀) (σᵣ ₁) (σᵣ ₂) (σᵣ ₃) ≡ true
  σᵣ-valid =
    ∧-true (ne ₀ ₁ Eq.refl) (∧-true (ne ₀ ₂ Eq.refl) (∧-true (ne ₁ ₂ Eq.refl)
      (∧-true (ne ₀ ₃ Eq.refl) (∧-true (ne ₁ ₃ Eq.refl) (ne ₂ ₃ Eq.refl)))))

  -- The canonical word with the same permutation.
  pairᵣ : CW
  pairᵣ = pairW (σᵣ ₀) (σᵣ ₁) (σᵣ ₂) (σᵣ ₃)

  flipsᵣ : CW
  flipsᵣ = flips4 (βᵣ ₀) (βᵣ ₁) (βᵣ ₂) (βᵣ ₃)

  canonW : CW
  canonW = pairᵣ • flipsᵣ

  private
    sel-σ : (v : Fin 4) → sel4 (σᵣ ₀) (σᵣ ₁) (σᵣ ₂) (σᵣ ₃) v ≡ σᵣ v
    sel-σ ₀ = Eq.refl
    sel-σ ₁ = Eq.refl
    sel-σ ₂ = Eq.refl
    sel-σ ₃ = Eq.refl

    sel-β : (v : Fin 4) → sel4b (βᵣ ₀) (βᵣ ₁) (βᵣ ₂) (βᵣ ₃) v ≡ βᵣ v
    sel-β ₀ = Eq.refl
    sel-β ₁ = Eq.refl
    sel-β ₂ = Eq.refl
    sel-β ₃ = Eq.refl

    -- The permutation at 2 v + e.
    at : (v : Fin 4) (e : Bool) → π (mk v e) ≡ mk (σᵣ v) (e xor βᵣ v)
    at v false = Eq.sym (mk-hl (π (mk v false)))
    at v true  = Eq.trans (Eq.cong π (Eq.sym (μ₀-mk v))) (h (mk v false))

    fl-at : (x : Fin 8) → act flipsᵣ x ≡ mk (half x) (low x xor βᵣ (half x))
    fl-at x = Eq.trans (flips-act (βᵣ ₀) (βᵣ ₁) (βᵣ ₂) (βᵣ ₃) x)
                (Eq.cong (λ z → mk (half x) (low x xor z)) (sel-β (half x)))

    pa-at : (v : Fin 4) (e : Bool) → act pairᵣ (mk v e) ≡ mk (σᵣ v) e
    pa-at v e = Eq.trans (pair-act (σᵣ ₀) (σᵣ ₁) (σᵣ ₂) (σᵣ ₃) σᵣ-valid (mk v e))
                  (Eq.cong₂ mk (Eq.trans (Eq.cong (sel4 (σᵣ ₀) (σᵣ ₁) (σᵣ ₂) (σᵣ ₃)) (half-mk v e)) (sel-σ v))
                               (low-mk v e))

  canonW-act : (x : Fin 8) → π x ≡ act canonW x
  canonW-act x =
    Eq.trans (Eq.cong π (Eq.sym (mk-hl x)))
      (Eq.trans (at (half x) (low x))
        (Eq.sym (Eq.trans (Eq.cong (act pairᵣ) (fl-at x)) (pa-at (half x) (low x xor βᵣ (half x))))))

  -- So the circuit of r commutes with G.
  W-core : Comm (Pc 0 r) G
  W-core = comm-≈ (perm-complete r canonW canonW-act)
    (comm-• (pair-comm (σᵣ ₀) (σᵣ ₁) (σᵣ ₂) (σᵣ ₃) σᵣ-valid) (flips-comm (βᵣ ₀) (βᵣ ₁) (βᵣ ₂) (βᵣ ₃)))

-- Frames of G with the same matching

-- The matching of G placed by p.
match : CW → Fin 8 → Fin 8
match p x = act p (μ₀ (act (inv p) x))

_==₈_ : Fin 8 → Fin 8 → Bool
x ==₈ y = (half x ==₄ half y) ∧ not (low x xor low y)

==₈-sound : (x y : Fin 8) → (x ==₈ y) ≡ true → x ≡ y
==₈-sound x y h = Eq.trans (Eq.sym (mk-hl x)) (Eq.trans (Eq.cong₂ mk hh ll) (mk-hl y))
  where
  hh : half x ≡ half y
  hh = ==₄-sound _ _ (∧ˡ h)
  xl : (a b : Bool) → not (a xor b) ≡ true → a ≡ b
  xl false false _ = Eq.refl
  xl true  true  _ = Eq.refl
  xl false true  ()
  xl true  false ()
  ll : low x ≡ low y
  ll = xl (low x) (low y) (∧ʳ {half x ==₄ half y} h)

all8 : (Fin 8 → Bool) → Bool
all8 f = f ₀ ∧ (f ₁ ∧ (f ₂ ∧ (f ₃ ∧ (f ₄ ∧ (f ₅ ∧ (f ₆ ∧ f ₇))))))

all8-sound : (f : Fin 8 → Bool) → all8 f ≡ true → ∀ x → f x ≡ true
all8-sound f h ₀ = ∧ˡ h
all8-sound f h ₁ = ∧ˡ (∧ʳ {f ₀} h)
all8-sound f h ₂ = ∧ˡ (∧ʳ {f ₁} (∧ʳ {f ₀} h))
all8-sound f h ₃ = ∧ˡ (∧ʳ {f ₂} (∧ʳ {f ₁} (∧ʳ {f ₀} h)))
all8-sound f h ₄ = ∧ˡ (∧ʳ {f ₃} (∧ʳ {f ₂} (∧ʳ {f ₁} (∧ʳ {f ₀} h))))
all8-sound f h ₅ = ∧ˡ (∧ʳ {f ₄} (∧ʳ {f ₃} (∧ʳ {f ₂} (∧ʳ {f ₁} (∧ʳ {f ₀} h)))))
all8-sound f h ₆ = ∧ˡ (∧ʳ {f ₅} (∧ʳ {f ₄} (∧ʳ {f ₃} (∧ʳ {f ₂} (∧ʳ {f ₁} (∧ʳ {f ₀} h))))))
all8-sound f h ₇ = ∧ʳ {f ₆} (∧ʳ {f ₅} (∧ʳ {f ₄} (∧ʳ {f ₃} (∧ʳ {f ₂} (∧ʳ {f ₁} (∧ʳ {f ₀} h))))))

-- The two matchings agree on the eight positions.
eqMatch : CW → CW → Bool
eqMatch p q = all8 (λ x → match p x ==₈ match q x)

eqMatch-sound : (p q : CW) → eqMatch p q ≡ true → ∀ x → match p x ≡ match q x
eqMatch-sound p q h x = ==₈-sound _ _ (all8-sound (λ y → match p y ==₈ match q y) h x)

match-InC : (p q : CW) → (∀ x → match p x ≡ match q x) → InC (inv q • p)
match-InC p q e x =
  Eq.trans (Eq.cong (act (inv q)) step) (act-inv q (μ₀ (act (inv q) (act p x))))
  where
  step : act p (μ₀ x) ≡ act q (μ₀ (act (inv q) (act p x)))
  step = Eq.trans (Eq.cong (λ z → act p (μ₀ z)) (Eq.sym (act-inv p x))) (e (act p x))

-- Two frames of G with the same matching are equal.
W′ : (p q : CW) → eqMatch p q ≡ true → 4 ⊢ Fr G p ≈ Fr G q
W′ p q e = Wc p q (W-core (inv q • p) (match-InC p q (eqMatch-sound p q e)))
