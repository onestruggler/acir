\section{Lessons}

\begin{frame}{Setoids, not quotients: the balance sheet}
\begin{columns}[T]
\begin{column}{0.47\textwidth}
\begin{tcolorbox}[colback=tinthot,colframe=hot,boxrule=0.6pt,arc=2pt,
  title=What it cost,fonttitle=\bfseries\small]
\small Every function out of a presented monoid carries a congruence proof
(bundled by the standard library's \aid{Injection}, \aid{RightInverse}, \ldots).
\end{tcolorbox}
\vspace{4pt}
\small A univalent quotient would turn $\approx$ into $\equiv$ --- but its eliminator
still demands a respect proof for every function out of it.
\end{column}
\begin{column}{0.49\textwidth}
\begin{win}[title=What it bought]
\small
\begin{itemize}
  \item normal forms, coset tables, sections \alert{run}: hundreds of obligations
        close by \aid{refl};
  \item \alert{maps into the syntax} --- sections, transversals, amalgamation
        tails --- need concrete representatives;
  \item plain Agda and the stdlib algebra hierarchy: \alert{upstreamable};
  \item derivations are data: lifting lemmas are inductions.
\end{itemize}
\end{win}
\end{column}
\end{columns}
\end{frame}

\begin{column}{0.50\textwidth}
\textbf{Limitations}
\begin{itemize}
  \small
  \item the amalgamation studies stop at \alert{syntax}: the isomorphism with the
        matrix groups is taken from the literature --- the rings and unitary
        matrices are not in the standard library;
  \item \alert{continuous parameters} (phase gates with real angles) have no finite
        coset structure: vocabulary yes, engine no;
  \item \alert{infinite-index} steps ($\mathbb{Z}_N\wr\mathbb{Z}$) are statable, but
        need induction instead of finitely many evaluations.
\end{itemize}
\end{column}
\end{columns}

\begin{columns}[T]
\begin{column}{0.46\textwidth}
\small \textbf{vs.\ abstract presentation theory}: no infrastructure to prove that a
\emph{given} relation set presents a \emph{given} group --- no verified coset
enumeration, no transversals.
\end{column}
\begin{column}{0.46\textwidth}
\small \textbf{vs.\ completion}: we never orient rules or prove confluence; a normal
form is any verified \emph{function} --- avoiding Squier's obstruction.
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Takeaways}
\begin{columns}[c]
\begin{column}{0.55\textwidth}
\begin{enumerate}
  \setlength{\itemsep}{8pt}
  \item \textbf{Completeness = soundness + unique normal form} --- one generic
        lemma; case studies only reason about finite data.
  \item \textbf{Coset tables are certificates}: finite data plus five equations,
        checked by coverage and closed by evaluation.
  \item \textbf{Normal forms grow one wire at a time} --- mixed-radix numerals over
        the chain the circuit layer hands us.
  \item \textbf{Constructions come in pairs}, against groups we added to the
        standard library.
\end{enumerate}
\end{column}
