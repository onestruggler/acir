\vspace{14pt}
\begin{idea}
\small The common thread: \alert{the same objects both compute and prove}. Normal forms,
coset tables and sections are structurally recursive programs that Agda's evaluator
runs; their correctness is checked once, generically.
\end{idea}
\end{frame}

\begin{frame}{\textbf{1}\; Presented monoids are inductive congruences}
\begin{columns}[c]
\begin{column}{0.42\textwidth}
A word is a syntax tree, \emph{not} a list:
\begin{code}
data Word (X : Set) : Set where
  [_]ʷ : X → Word X
  ε    : Word X
  _•_  : Word X → Word X → Word X
\end{code}
\begin{code}[hide]
infixr 7 _•_
infix 4 _≈_ _===_ _≈ₙ_ _≈₂_

WRel : Set → Set₁
WRel X = Rel (Word X) 0ℓ

postulate
  w v u w' v' : Word X
  _===_       : WRel X
\end{code}
\vspace{2pt}
\footnotesize
\begin{itemize}
  \item terms mirror pen-and-paper calculations;
  \item associativity is a \emph{rule}, discharged by a solver
        (\aid{by-assoc}: flatten, compare by \aid{refl});
  \item groups are monoids whose generators have left inverses
        (\aid{Grouplike}).
\end{itemize}
\end{column}
\begin{column}{0.54\textwidth}
{\small \aid{\_===\_}: the axioms; \aid{\_≈\_}: the congruence they generate.}
\renewcommand{\AgdaCodeStyle}{\footnotesize}
\begin{code}
data _≈_ : WRel X where
  refl  : w ≈ w
  sym   : w ≈ v → v ≈ w
  trans : w ≈ v → v ≈ u → w ≈ u
  cong  : w ≈ w' → v ≈ v' → w • v ≈ w' • v'
  assoc      : (w • v) • u ≈ w • (v • u)
  left-unit  : ε • w ≈ w
  right-unit : w • ε ≈ w
  axiom      : w === v → w ≈ v
\end{code}
\begin{win}
\small Derivations are \alert{data}: lifting a whole congruence one wire up, or
across a product, is an induction over these eight constructors.
\end{win}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{\textbf{2}\; Normal forms are standard-library bundles}
\begin{code}[hide]
postulate
  |NF|       : Set
  nf         : Word X → |NF|
  _≈ₙ_       : |NF| → |NF| → Set
  NormalForm : Set
  inv-nf     : |NF| → Word X
  Sem        : Set
  ⟦_⟧        : Word X → Sem
  _≈₂_       : Sem → Sem → Set
\end{code}
\begin{columns}[c]
\begin{column}{0.50\textwidth}
From the \alert{word setoid} to a carrier $\mathit{NF}$:
\vspace{4pt}

\begin{tabular}{@{\quad}l@{\;\;=\;\;}l@{}}
\aid{NormalFormInjective} & \aid{Injection}\\
\aid{NormalForm}          & \aid{RightInverse}\\
\aid{BijectiveNormalForm} & \aid{Bijection}
\end{tabular}
\vspace{6pt}

No new theory, just names for roles. An \aid{Injection} already gives
\begin{code}
by-equal-nf : nf w ≈ₙ nf v → w ≈ v
≈-dec : Decidable _≈ₙ_ → Decidable _≈_
\end{code}
\begin{code}[hide]
by-equal-nf = proof-elided
≈-dec = proof-elided
\end{code}
{\small prove equations by \alert{computing} normal forms; decide the word problem.}
\end{column}

\vspace{6pt}
\begin{idea}[title={Constructively, the section is data}]
\small Classically an injective \aid{nf} has a section for free (choose a
preimage). In Agda it is an \emph{algorithm} --- the extra field of
\aid{RightInverse} --- and it is what completeness consumes.
\end{idea}

\begin{frame}{Completeness, once and for all}
A normal form is \alert{unique} for a semantics \aid{⟦\_⟧} (into any setoid) when
canonical representatives with equal meanings are equal:
\begin{code}
record UniqueNormalForm (normalForm : NormalForm) : Set (c ⊔ d) where
  field unique : ∀ {u v : |NF|} → ⟦ inv-nf u ⟧ ≈₂ ⟦ inv-nf v ⟧ → u ≈ₙ v
by-normalization : {normalForm : NormalForm} → UniqueNormalForm normalForm →
  Congruent _≈_ _≈₂_ ⟦_⟧ → Injective _≈_ _≈₂_ ⟦_⟧
\end{code}
\begin{code}[hide]
by-normalization = proof-elided
\end{code}
\vspace{4pt}
\begin{columns}[c]
\begin{column}{0.55\textwidth}
\begin{win}
\textbf{soundness + unique normal form $\Rightarrow$ completeness.}\\[2pt]
\small Five lines. Every case study proves only \aid{unique} --- a statement
about \alert{normal forms}, i.e.\ finite data, never about arbitrary words.
\end{win}
\end{column}
\begin{column}{0.42\textwidth}
\small
\begin{itemize}
  \item \aid{by-completeness}: the converse, given an exact section --- how
        uniqueness travels through products;
  \item with surjectivity, the records assemble a group isomorphism:
        \aid{\_IsPresentationOf\_}.
\end{itemize}
\end{column}
\end{columns}
\end{frame}
