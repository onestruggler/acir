\section{Verified coset tables}

\begin{frame}{\textbf{3}\; Certificates are finite: the Reidemeister--Schreier engine}
\small $\Gamma$ presents a subgroup $H$ of the group $G$ presented by $\Delta$.
A \alert{claimed} coset table --- cosets $C$ with identity $I$, a section
$[\_] : C\to\aid{Word}\;Y$, and a table $h$ that pushes a letter past a coset and
emits a word of $H$ --- is \alert{verified} against five equations; the search
that found it stays outside.
\vspace{4pt}

\renewcommand{\AgdaCodeStyle}{\scriptsize}
\begin{code}
module Transfer
  (h=⁻¹f-gen : ∀ (x : X) → ([ x ]ʷ , I) ~ ((h ᵗ) I (f x)))                         -- 1
  (h-wd-ax : ∀ (c : C){u t : Word Y} → u ===₂ t → ((h ᵗ) c u) ~ ((h ᵗ) c t))       -- 2
  (f-wd-ax : ∀ {w v} → w ===₁ v → (f ʷ) w ≈₂ (f ʷ) v)                              -- 3
  ([I]≈ε : [ I ] ≈₂ ε)                                                             -- 4
  (h=ract : ∀ c b → let (b' , c') = h c b in [ c ] • [ b ]ʷ ≈₂ (f ʷ) b' • [ c' ])  -- 5
\end{code}

\begin{frame}{One more digit per level: normal-form transport and towers}
\begin{columns}[c]
\begin{column}{0.50\textwidth}
From the five equations, proved once:
\[
\begin{array}{@{}l@{}}
  \aid{nf-transp} : \aid{NormalForm}\;\Gamma\;\mathit{NF}\\
  \qquad\qquad\to \aid{NormalForm}\;\Delta\;(\mathit{NF}\times C)
\end{array}
\]
\small a normal form for the subgroup gives one for the group, \alert{with one more
coset digit}. \aid{PackedCosetTable} makes a verified level a first-class value;
\aid{CosetTower} stacks them:
\[
  \mathit{NF}_0 \times C_1 \times C_2 \times \cdots \times C_k
\]
--- the classical stabilizer-chain normal form, read as a \alert{mixed-radix
numeral} (Schreier--Sims), with syntactically chosen chains: for circuits,
\emph{one level per wire}.
\end{column}
