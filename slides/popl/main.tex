%% \swaps{n}{p1,p2,...}: n wires (wire 0 at the bottom), a swap of the
%% wires p, p+1 at each successive step; `x' leaves the step empty.
\newcommand{\swaps}[3][0.3]{%
  \begin{tikzpicture}[x=#1cm,y=#1cm,line width=0.7pt,draw=wire,
      baseline={([yshift=-0.5ex]current bounding box.center)}]
    \def\gap{x}\def\len{0}%
    \foreach \p [count=\i] in {#3} {\xdef\len{\i}}%
    \pgfmathsetmacro{\L}{2*\len+1}%
    \foreach \w in {1,...,#2} {\draw (0,{(\w-1)*1.6}) -- (\L,{(\w-1)*1.6});}
    \foreach \p [count=\i] in {#3} {%
      \ifx\p\gap\else
      \pgfmathsetmacro{\xa}{2*\i-1.45}\pgfmathsetmacro{\xb}{2*\i+0.45}%
      \pgfmathsetmacro{\ya}{\p*1.6}\pgfmathsetmacro{\yb}{(\p+1)*1.6}%
      \fill[white] (\xa,\ya-0.05) rectangle (\xb,\yb+0.05);
      \draw (\xa,\ya) -- (\xb,\yb); \draw (\xa,\yb) -- (\xb,\ya);
      \fi}
  \end{tikzpicture}}
