conj-Ex-H↑ : Ex • H ↑ • Ex ≈ H ↓
conj-Ex-H↑ = begin
  Ex • H ↑ • Ex    ≈⟨ sym assoc ⟩
  (Ex • H ↑) • Ex  ≈⟨ cleft axiom semi-Ex-H↑ ⟩
  (H ↓ • Ex) • Ex  ≈⟨ assoc ⟩
  H ↓ • Ex • Ex    ≈⟨ cright axiom order-Ex ⟩
  H ↓ • ε          ≈⟨ right-unit ⟩
  H ↓ ∎
