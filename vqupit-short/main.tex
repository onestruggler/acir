\newcommand{\fitfig}[2][\figwidth]{%
  \sbox{\figbox}{\tikzfig{figures/#2}}%
  \ifdim\wd\figbox>#1
    \sbox{\figbox}{\resizebox{#1}{!}{\usebox{\figbox}}}%
  \fi
  $\vcenter{\hbox{\usebox{\figbox}}}$}
