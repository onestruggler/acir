\section{Introduction}

\begin{frame}{From qubit to qudit Clifford gates}
  Fix an odd prime $p$ and let $\omega = e^{2\pi i/p}$.
  The Clifford group $\mathcal{C}_n$ is generated by:
  \vfill
  \begin{center}
  \begin{tabular}{lll}
    \toprule
      & qubit ($p = 2$) & qudit (odd prime $p$) \\
    \midrule
    $H$  & $\frac{1}{\sqrt2}\begin{pmatrix}1&1\\1&-1\end{pmatrix}$
         & $\ket{j} \mapsto \tfrac{1}{\sqrt p}\sum_k \omega^{jk}\ket{k}$ \\[2ex]
    $S$  & $\begin{pmatrix}1&0\\0&i\end{pmatrix}$
         & $\ket{j} \mapsto \omega^{j(j-1)/2}\,\ket{j}$ \\[2ex]
    $CZ$ & $\mathrm{diag}(1,1,1,-1)$
         & $\ket{j}\ket{l} \mapsto \omega^{jl}\,\ket{j}\ket{l}$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vfill Note the $/2$ in the qudit $S$ is the inverse of $2$ in
  $\mathbb{Z}_p$. A global phase can be added to the defition of $H$
  to facilitate computation.
\end{frame}

\section{Main results}

\begin{frame}{Main result 1: a presentation of the qudit Clifford group}
  The paper's Figure~1 rule set, mod scalars, \alert{presents} the
  projective Clifford group
  $\mathcal{C}_n/\langle\omega\rangle \;\cong\;
   \mathrm{Pauli}_n \rtimes \mathrm{Sp}(2n, \mathbb{Z}_p)$:

\begin{code}[hide]
module Main-result-1 where
  open PapSyn.Clifford-Relations renaming (_QRel,_===_ to _CRel,_===_)
  open SDPres.Semidirect using (Pauli⋊Sp)
\end{code}
\begin{code}
  Theorem-1 : ∀ (n : ℕ) → (n CRel,_===_) IsPresentationOf (Pauli⋊Sp n)
  Theorem-1 = PapPres.presentation
\end{code}
  \vfill
  \begin{itemize}
    \item \AgdaDatatype{\_CRel,\_===\_} is the 16-relation rule set,
          shown on the next slides.
    \item The theorem holds \emph{at every width $n$}, for \emph{every}
          odd prime $p$ and every primitive root $g$ --- the whole deck
          is parameterised by them.
  \end{itemize}
\end{frame}

\begin{frame}{The 16 relations, in Agda (one qudit)}
\begin{code}[hide]
module The-sixteen-relations where
  open PapSyn using (Gen ; S ; S⁻¹ ; H ; CZ ; Ex ; CX ; CZ02 ; S^ ; CZ^ ; _↑ ; _↓)
  open PapSyn.Clifford-Relations using (M₋₁ ; XM ; XMg ; XMg^ ; R ; R^)
  Mg = XMg
  Mg^ = XMg^
  M = XM
  private variable n : ℕ
  infix 4 _CRel,_===_
\end{code}
\begin{code}
  data _CRel,_===_ : (n : ℕ) → WRel (Gen n) where
    order-S      :        (₁₊ n) CRel,  S ^ p === ε
    order-H      :        (₁₊ n) CRel,  H ^ 2 === M₋₁
    M-power      : ∀ k →  (₁₊ n) CRel,  Mg^ k === M (g^ k)
    semi-MR      :        (₁₊ n) CRel,  R • Mg === Mg • R^ (g * g)
    order-SH     :        (₁₊ n) CRel,  (S • H) ^ 3 === ε
    comm-HHSHHS  :        (₁₊ n) CRel,  H • H • S • H • H • S === S • H • H • S • H • H
\end{code}
  \vfill
  \begin{itemize}
    \item $M_x$ is the multiplier $\ket{j} \mapsto \ket{xj}$, spelled as the
          word $R^{x^{-1}}\, H\, R^x H\, R^{x^{-1}}\, H$ with $R = S \cdot Z^{1/2}$.
    \item One-qudit relations talk about the powers of a \alert{fixed
          primitive root} $g$ --- this is where odd primes differ from
          the qubit case.
  \end{itemize}
\end{frame}

\begin{frame}{The 16 relations, in Agda (two qudits)}
\begin{code}
    order-CZ     :        (₂₊ n) CRel,  CZ ^ p === ε
    order-Ex     :        (₂₊ n) CRel,  Ex ^ 2 === ε
    comm-CZ-S↑   :        (₂₊ n) CRel,  CZ • S ↑ === S ↑ • CZ
    semi-M↑CZ    :        (₂₊ n) CRel,  CZ • Mg ↑ === Mg ↑ • CZ^ g
    semi-Ex-S↑   :        (₂₊ n) CRel,  Ex • S ↑ === S ↓ • Ex
    semi-Ex-H↑   :        (₂₊ n) CRel,  Ex • H ↑ === H ↓ • Ex
    blake-c12    :        (₂₊ n) CRel,  (S ^ p-1) ↑ • (S ^ p-1) ↓ • CX ^ p-1 • S ↓ • CX === CZ
\end{code}
  \vfill

    \item \AgdaInductiveConstructor{blake-c12} converts between the two

  \end{itemize}
\end{frame}

\begin{frame}{The 16 relations, in Agda (three qudits)}
\begin{code}
    yang-baxter  :        (₃₊ n) CRel,  Ex ↑ • Ex ↓ • Ex ↑ === Ex ↓ • Ex ↑ • Ex ↓
    cz-slide     :        (₃₊ n) CRel,  Ex ↓ • Ex ↑ • CZ === CZ ↑ • Ex ↓ • Ex ↑
    semi-CX↑-CZ↓ :        (₃₊ n) CRel,  CZ ↓ • CX ↑ === CZ02 • CX ↑ • CZ ↓
\end{code}

    \item Only \alert{these 16} are axioms.  $XZ = ZX\,(\text{mod }

    \item Plus, for every gate set, the same three monoidal structural
      rules: congruence under $\uparrow$, gates on disjoint wires
      commute, and scalars commutes with every gate.
  \end{itemize}
\end{frame}

\begin{frame}{What ``presents'' means}
  \begin{code}[hide]
module What-presents-means {a ℓ : Level} {X : Set} where
\end{code}
\begin{code}
  record _Presents_ (_===_ : WRel X) (G : Group a ℓ) : Set (a ⊔ ℓ) where
    field gl : Grouplike _===_
    module GL = Group-Lemmas _===_ gl
    open GroupMorphisms (Group.rawGroup GL.•-ε-group) (Group.rawGroup G)
    field
      ⟦_⟧  : Word X → Group.Carrier G
      iso  : IsGroupIsomorphism ⟦_⟧
\end{code}
  \vfill
  \begin{center}
  \begin{tikzpicture}[scale=1.0]
    \node (a) at (0,0) {$\mathsf{Cir}\;n \,/\, \approx_{\text{16 rules}}$};
    \node (b) at (7,0) {$\mathrm{Pauli}_n \rtimes \mathrm{Sp}(2n,\mathbb{Z}_p)
                         \;\cong\; \mathcal{C}_n/\langle\omega\rangle$};
    \draw[->, thick] (a) -- node[above]{{\AgdaCodeFont ⟦\_⟧}}
                          node[below]{$\cong$} (b);
  \end{tikzpicture}
  \end{center}
  \begin{itemize}
    \item \emph{Soundness}: {\AgdaCodeFont ⟦\_⟧} is well defined ---
          each rule holds semantically.
    \item \emph{Completeness}: {\AgdaCodeFont ⟦\_⟧} is injective ---
          semantically equal circuits are related by the rules.
  \end{itemize}
\end{frame}

\section{Method}

\begin{frame}{Divide and conquer}
  Factor the group; present the factors; compose the presentations.
  \vfill
  \begin{center}
  \begin{tikzpicture}[scale=0.9]
    \node at (0,1.5) {$1 \to \mathrm{Pauli}_n \to
      \mathcal{C}_n/\langle\omega\rangle \to \mathrm{Sp}(2n,\mathbb{Z}_p)
      \to 1$};
    \node[anchor=west] at (7.2,1.5) {\small semidirect: it splits};
    \node at (0,0.4) {$1 \to \langle\omega\rangle \to
      \mathcal{C}_n \to \mathcal{C}_n/\langle\omega\rangle \to 1$};
    \node[anchor=west] at (7.2,0.4) {\small central extension};
  \end{tikzpicture}
  \end{center}
  \vfill
  \begin{itemize}
    \item Doesn't this just move the difficulty around?  \alert{No}: the
          symplectic factor is where the real rewriting happens, and
          keeping the Pauli part out of it pays off everywhere.
    \item Prior work (qubit: Selinger; qutrit: Li et al.) worked with
      the whole Clifford group at once --- the derivations drag a
      ``messy'' Pauli correction through every step.
  \end{itemize}
\end{frame}

\begin{frame}{Coset representatives, and the normal form}
  Representatives $C_2$ of $S_3/S_2$ (and of $S_2/S_1$, $S_1/S_0$):
  the staircases \vfill
  \begin{center}
  \begin{tabular}{ccc@{\qquad\qquad}cc@{\qquad\qquad}c}
    \ufig[0.8]{c2-0} & \ufig[0.8]{c2-1} &

    $c^{(0)}_0$
  \end{tabular}

    \begin{column}{0.68\textwidth}

          \cdot 2 \cdot 1$.

\begin{frame}{Rewriting to the normal form: push a generator through}
  Multiply a normal form by a generator; push it through the staircase:
  \begin{center}\small
  \begin{tabular}{r@{\qquad}l}
    \ufig[0.62]{push-commute} & disjoint wires: commute past \\
    \ufig[0.62]{push-pass}    & Yang--Baxter: pass through, shift up \\
    \ufig[0.62]{push-absorb}  & $\sigma^2 = e$: staircase shrinks \\
    \ufig[0.62]{push-extend}  & staircase grows \\
  \end{tabular}
  \end{center}
  Each case is a \emph{relation}; collecting them all gives a rewrite
  system that normalises every circuit.
\end{frame}

\begin{frame}{From $S_n$ to $\mathrm{Sp}(2n,\mathbb{Z}_p)$: change the action}
  The same framework, three ways of looking at the wires:
  \vfill
  \begin{enumerate}
    \item $S_n$ acts on $\{0, \dots, n-1\}$: a wire is a \emph{position};
          $c$ sends $f^{-1}(0)$ to $0$.
    \item $S_n$ acts on $\mathsf{Vec}\;\mathsf{Bit}\;n$: a wire is a
          \emph{bit}; $f^{-1}c$ fixes the linear subspace of the bottom wire.
    \item Adding $H, S, CZ$: the group acts on
          $\mathsf{Vec}\;\mathsf{Pauli}_1\;n$ --- a $2n$-dimensional
          \alert{symplectic} $\mathbb{Z}_p$-vector space.  A wire is a
          $\mathrm{Pauli}_1$; enlarging the coset layer from staircases to
          $\mathit{ML}$ boxes, $f^{-1}c$ fixes the symplectic subspace
          of the bottom wire.
  \end{enumerate}
  \vfill
  The framework applies to \emph{any} $\mathbb{N}$-indexed family of
  groups with such a layered structure
  (\texttt{Circuit.CosetNF}, \texttt{Circuit.Uniqueness}).
\end{frame}

\begin{frame}{Composing: the semidirect product}
  Semidirect product: relations of both factors, plus twisted commuting,
  $h\,n' = ({}^{h}n')\,h$, so
  $(n , h) \cdot (n' , h') = (n \cdot {}^{h}n'\, ,\; h\,h')$:

\begin{code}[hide]
module Semidirect-glue where
  private variable n : ℕ
\end{code}
\begin{code}
  conj : Sym.Gen n → XZ.Gen n → Word (XZ.Gen n)
  conj Sym.H-gen   XZ.X-gen         = XZ.Z
  conj Sym.H-gen   XZ.Z-gen         = XZ.X ^ p-1
  conj Sym.S-gen   XZ.X-gen         = XZ.X • XZ.Z
  conj Sym.S-gen   XZ.Z-gen         = XZ.Z
  conj Sym.CZ-gen  XZ.X-gen         = XZ.X • XZ.Z XZ.↑
  conj Sym.CZ-gen  (XZ.X-gen XZ.↥)  = XZ.X XZ.↑ • XZ.Z
\end{code}
\begin{code}[hide]
  conj Sym.CZ-gen  XZ.Z-gen          = XZ.Z
  conj Sym.CZ-gen  (XZ.Z-gen XZ.↥)   = XZ.Z XZ.↑
  conj Sym.H-gen   (xz XZ.↥)         = [ xz ]ʷ XZ.↑
  conj Sym.S-gen   (xz XZ.↥)         = [ xz ]ʷ XZ.↑
  conj Sym.CZ-gen  (xz XZ.↥ XZ.↥)    = [ xz ]ʷ XZ.↑ XZ.↑
  conj (sym Sym.↥) XZ.X-gen          = XZ.X
  conj (sym Sym.↥) XZ.Z-gen          = XZ.Z
  conj (sym Sym.↥) (xz XZ.↥)         = conj sym xz XZ.↑
\end{code}
\begin{code}
  Pauli⋊Symplectic : (n : ℕ) → WRel (XZ.Gen n ⊎ Sym.Gen n)
  Pauli⋊Symplectic n = XZ._QRel,_===_ n ⋄ Sim._QRel,_===_ n ⋄ ConjRelʷ conj
\end{code}
  \vfill
  The projection to the symplectic factor is a homomorphism, so
  symplectic circuits and their relations \alert{lift} unchanged.
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item \alert{Main result 1}: presentations by generators and relations
          of the qudit Clifford operators (16 relations, $\leq 3$ qudits)
          and of the symplectic operators, for all odd primes at once.
    \item \alert{Main result 2}: a unique normal form for qudit Clifford
          and symplectic operators.
    \item Both are \alert{checked in Agda} --- except for the two classical
          glue facts: that projective Clifford is
          $\mathrm{Pauli} \rtimes \mathrm{Sp}$, and that Clifford is a
          central extension of projective Clifford by the scalars.
  \end{itemize}
